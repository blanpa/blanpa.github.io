{{- /* Monthly npm downloads for a project, from data/npm_stats.yml (baked in by the deploy workflow). Renders nothing when the page declares no `npm:` package or the stats file has no entry for it, so a new project or a failed refresh degrades to silence rather than to a zero. Param: the project Page. */ -}} {{- $pkg := .Params.npm -}} {{- with $pkg -}} {{- $downloads := index (site.Data.npm_stats.packages | default dict) . -}} {{- $release := index (site.Data.npm_stats.versions | default dict) . -}} {{- if or $downloads $release -}} {{- with $downloads }} {{ lang.FormatNumber 0 . }}/mo {{- end }} {{- with $release }} {{- $when := "" -}} {{- with .released }}{{ $when = printf " · %s" (time.Format "Jan 2006" .) }}{{ end -}} v{{ .version }}{{ $when }} {{- end }} {{- end -}} {{- end -}}
