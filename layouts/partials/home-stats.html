{{- /* The numbers, as a band under the hero: downloads, packages, the Node-RED library rating and the article count. Every figure is baked in at build time — npm from data/npm_stats.yml, the rating from data/nodered_ratings.yml (both refreshed by the deploy workflow), the article count from the content — so they are real on first paint with no client-side fetch. */ -}} {{- $total := site.Data.npm_stats.total_last_month | default 0 -}} {{- $count := len site.Data.npm_packages -}} {{- $articles := len (where site.RegularPages "Section" "blog") -}} {{- /* Weighted by votes, so one 5-star vote on a new package does not count as much as seven on an established one. */ -}} {{- $votes := 0 }}{{ $points := 0.0 }} {{- range site.Data.nodered_ratings.packages | default dict }} {{- if .count }} {{- $votes = add $votes .count }} {{- $points = add $points (mul (float .score) .count) }} {{- end }} {{- end }}
{{ lang.FormatNumber 0 $total }} npm downloads a month
{{ $count }} open-source packages
{{- if gt $votes 0 }}
{{ lang.FormatNumber 1 (div $points $votes) }}/5 Node-RED library rating, {{ $votes }} votes
{{- end }}
