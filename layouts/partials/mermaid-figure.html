{{- /* Shared markup for both authoring paths: ```mermaid fences (via the codeblock render hook) and the {{< mermaid >}} shortcode. Two rendering paths, in this order of preference: 1. STATIC — tools/render-diagrams.py has pre-rendered this diagram to assets/diagrams/-{light,dark}.svg. Both variants are inlined and CSS shows the one matching the theme. No JavaScript involved: the reader gets the finished picture, the diagram survives with JS disabled, and the 3.2 MB mermaid runtime is never fetched. 2. CLIENT-SIDE — no SVG for this source yet (a diagram edited but not re-rendered, or authoring before the script has run). Falls back to the previous behaviour: Blowfish's appearance.js renders every 
 and re-renders it on theme switch, so the
          
 class list is load-bearing. The runtime loader is emitted once
          per page and ui.js fetches the bundle when a diagram nears the
          viewport. (This can't live in extend-footer.html: Blowfish renders
          that partial through partialCached with no key, i.e. once for the
          whole site.)

       The key must match what render-diagrams.py computes — see normalise()
       there: CRLF folded, then stripped. tests/test_links_and_diagrams.py
       fails when a diagram has no SVG, which is what stops path 2 from
       becoming the silent default again.

       Params: code (the diagram source), Page. */ -}}

{{- $code := replace .code "\r\n" "\n" -}}
{{- $key := substr (sha1 (trim $code " \n\t")) 0 16 -}}
{{- $light := resources.Get (printf "diagrams/%s-light.svg" $key) -}}
{{- $dark := resources.Get (printf "diagrams/%s-dark.svg" $key) -}}

{{- if and $light $dark -}}

  {{- /* mmdc gives every SVG root id="my-svg" and scopes the diagram's own
         stylesheet to it — "#my-svg .node rect{fill:...}". Both variants are
         inlined into the same document, so left alone those two style blocks
         match BOTH SVGs and whichever sits later in the DOM (dark) wins for
         both: in the light theme the diagram drew near-black boxes on a white
         page. Rewriting the id scopes each stylesheet back to its own SVG.
         The ordinal separates two copies of one diagram on a page — the key
         is the source hash, so it would not. Every "my-svg" in the file is an
         id or a url(#…) pointing at one, and the string appears in no diagram
         source, so replacing it wholesale is safe. */ -}}
  {{- $n := add (int (.Page.Store.Get "mermaidSeq")) 1 -}}
  {{- .Page.Store.Set "mermaidSeq" $n -}}
  {{- $id := printf "mmd%d-%s" $n $key -}}

  

    
{{ replace $light.Content "my-svg" (printf "%s-l" $id) | safeHTML }}

    
{{ replace $dark.Content "my-svg" (printf "%s-d" $id) | safeHTML }}

    
      
      Click to zoom
    
  


{{- else -}}

  {{- $first := not (.Page.Store.Get "hasMermaid") -}}
  {{- .Page.Store.Set "hasMermaid" true -}}
  {{- if $first -}}
    {{- with resources.Get "lib/mermaid/mermaid.min.js" -}}
      {{- $config := resources.Get "js/mermaid.js" | resources.Minify -}}
      {{- $bundle := slice . $config | resources.Concat "js/mermaid.bundle.js" | resources.Fingerprint (site.Params.fingerprintAlgorithm | default "sha512") -}}
      

    {{- end -}}
  {{- end -}}
  

    
{{ .code | safeHTML }}

    
      
      Click to zoom
    
  


{{- end -}}
